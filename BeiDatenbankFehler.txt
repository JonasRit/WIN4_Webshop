-- Tabellen droppen (falls kaputte Reste da sind)
DROP TABLE IF EXISTS warenkorb_produkt;
DROP TABLE IF EXISTS warenkorb;
DROP TABLE IF EXISTS produkt_produktkategorie;
DROP TABLE IF EXISTS produkt;
DROP TABLE IF EXISTS produktkategorie;
DROP TABLE IF EXISTS kunde;

-- Kunden
CREATE TABLE kunde (
    id INT AUTO_INCREMENT PRIMARY KEY,
    benutzername VARCHAR(255) NOT NULL,
    passwort VARCHAR(255) NOT NULL,
    vorname VARCHAR(255),
    name VARCHAR(255),
    anschrift VARCHAR(255),
    email VARCHAR(255)
);

-- Kategorien (self-referencing für Unterkategorien)
CREATE TABLE produktkategorie (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    parent_id INT DEFAULT NULL,
    FOREIGN KEY (parent_id) REFERENCES produktkategorie(id)
);

-- Produkte
CREATE TABLE produkt (
    id INT AUTO_INCREMENT PRIMARY KEY,
    bezeichnung VARCHAR(255) NOT NULL,
    beschreibung TEXT,
    preis DECIMAL(10,2),
    bild VARCHAR(255)
);

-- Zuordnung Produkt <-> Kategorie
CREATE TABLE produkt_produktkategorie (
    produkt_id INT,
    produktkategorie_id INT,
    PRIMARY KEY (produkt_id, produktkategorie_id),
    FOREIGN KEY (produkt_id) REFERENCES produkt(id),
    FOREIGN KEY (produktkategorie_id) REFERENCES produktkategorie(id)
);

-- Warenkorb
CREATE TABLE warenkorb (
    id INT AUTO_INCREMENT PRIMARY KEY,
    kunde_id INT,
    FOREIGN KEY (kunde_id) REFERENCES kunde(id)
);

-- Warenkorb-Produkte
CREATE TABLE warenkorb_produkt (
    warenkorb_id INT,
    produkt_id INT,
    PRIMARY KEY (warenkorb_id, produkt_id),
    FOREIGN KEY (warenkorb_id) REFERENCES warenkorb(id),
    FOREIGN KEY (produkt_id) REFERENCES produkt(id)
);
